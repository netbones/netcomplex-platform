  <files>src/lib/tenant/with-tenant.ts</files>
  <action>
    Create src/lib/tenant/with-tenant.ts with the following:
    
    ```typescript
    import { headers } from 'next/headers';
    
    export async function withTenant() {
      const headersList = headers();
      const tenantId = headersList.get('x-tenant-id');
      const tenantSlug = headersList.get('x-tenant-slug');
    
      if (!tenantId || !tenantSlug) {
        throw new Error('Tenant not resolved — middleware must run first');
      }
    
      return { tenantId, tenantSlug };
    }
    
    export async function withTenantOptional(): Promise<{ tenantId?: string; tenantSlug?: string }> {
      const headersList = headers();
      return {
        tenantId: headersList.get('x-tenant-id') ?? undefined,
        tenantSlug: headersList.get('x-tenant-slug') ?? undefined,
      };
    }
    ```
    
    This helper must be called at the top of every API route handler.
    It throws if tenant headers are not present (middleware failure).
  </action>
  <verify>ls -la src/lib/tenant/with-tenant.ts</verify>
