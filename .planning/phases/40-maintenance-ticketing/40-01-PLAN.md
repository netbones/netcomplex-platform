  id          String                @id
  tenantId    String
  name        String
  trade       String                // PLUMBING, ELECTRICAL, HVAC, LANDSCAPING, GENERAL
  contactName String?
  isActive    Boolean               @default(true)

  @@index([tenantId])
}

  isActive    Boolean               @default(true)
