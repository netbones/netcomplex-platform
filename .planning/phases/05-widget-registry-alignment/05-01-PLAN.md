```typescript
// src/widgets/dashboard/model/registry.ts

import type { WidgetManifest } from './types';
import type { WidgetRegistryEntry, WidgetComponent } from './registry';

class WidgetRegistry {
  private manifests = new Map<string, WidgetManifest>();

  register(manifest: WidgetManifest): void {
    if (this.manifests.has(manifest.id)) {
      console.warn(`[Registry] Overwriting widget: ${manifest.id}`);
    }
    this.manifests.set(manifest.id, manifest);
  }

  resolve(id: string): WidgetManifest | undefined {
    return this.manifests.get(id);
  }

  list(): WidgetManifest[] {
    return [...this.manifests.values()];
  }

  listByCategory(category: string): WidgetManifest[] {
    return this.list().filter(m => m.category === category);
  }

  listPremium(): WidgetManifest[] {
    return this.list().filter(m => m.premium);
  }

  // Filter by feature flags and role
  listForContext(enabledFeatures: string[], userRole: string): WidgetManifest[] {
    return this.list().filter(m => {
      if (m.featureFlag && !enabledFeatures.includes(m.featureFlag)) return false;
      if (m.permissions && !m.permissions.includes(userRole as any)) return false;
      return true;
    });
  }
}

export const registry = new WidgetRegistry();

// BACKWARD-COMPATIBLE EXPORTS — keep existing API working
export function getWidgetComponent(id: string): WidgetComponent | undefined {
  return registry.resolve(id)?.component;
}

export function getWidgetMetadata(id: string): WidgetRegistryEntry | undefined {
  const m = registry.resolve(id);
  if (!m) return undefined;
  // Map to existing shape for admin UI
  return {
    id: m.id,
    name: m.name,
    description: m.description,
    featureFlag: m.featureFlag,
    premium: m.premium,
    category: m.category,
  };
}

export function hasWidget(id: string): boolean {
  return registry.resolve(id) !== undefined;
}

export function getAllWidgets(): WidgetRegistryEntry[] {
  return registry.list().map(m => ({
    id: m.id,
    name: m.name,
    description: m.description,
    featureFlag: m.featureFlag,
    premium: m.premium,
    category: m.category,
  }));
}

export function getWidgetsByCategory(category: string): WidgetRegistryEntry[] {
  return registry.listByCategory(category).map(m => ({
    id: m.id,
    name: m.name,
    description: m.description,
    featureFlag: m.featureFlag,
    premium: m.premium,
    category: m.category,
  }));
}

export function getPremiumWidgets(): WidgetRegistryEntry[] {
  return registry.listPremium().map(m => ({
    id: m.id,
    name: m.name,
    description: m.description,
    featureFlag: m.featureFlag,
    premium: m.premium,
    category: m.category,
  }));
}
```

Key points:

- Use Map for O(1) lookups (existing object was O(n) for has/get)
- Add listForContext() for feature+role filtering
- ALL existing helper functions still work (backward compat)
- Registry auto-registers on import from widgets.ts
