import { MailerSend, EmailParams, Sender, Recipient } from 'mailersend';

const API_KEY = process.env.MAILERSEND_API_KEY;
const FROM_EMAIL = process.env.MAILERSEND_FROM_EMAIL || 'noreply@soralia.co.za';
const FROM_NAME = process.env.MAILERSEND_FROM_NAME || 'Soralia Village';

const mailerSend = new MailerSend({
  apiKey: API_KEY,
});

export const sentFrom = new Sender(FROM_EMAIL, FROM_NAME);

export async function sendEmail({
  to,
  subject,
  html,
  text,
  replyTo,
}: {
  to: string | string[];
  subject: string;
  html: string;
  text?: string;
  replyTo?: string;
}) {
  const recipients = Array.isArray(to)
    ? to.map(email => new Recipient(email))
    : [new Recipient(to)];

  const emailParams = new EmailParams()
    .setFrom(sentFrom)
    .setTo(recipients)
    .setSubject(subject)
    .setHtml(html)
    .setText(text || html.replace(/<[^>]*>/g, ''));

  if (replyTo) {
    emailParams.setReplyTo(new Sender(replyTo));
  }

  return mailerSend.email.send(emailParams);
}

export const templates = {
  welcome: {
    subject: 'Welcome to Soralia Village!',
    getHtml: (name: string) => `
      <h1>Welcome to Soralia Village, ${name}!</h1>
      <p>Your account has been created successfully.</p>
      <p>You can now access the community portal and connect with your neighbors.</p>
      <p>Login at: https://soralia.co.za/login</p>
    `,
  },

  passwordReset: {
    subject: 'Reset your Soralia Village password',
    getHtml: (resetUrl: string) => `
      <h1>Password Reset Request</h1>
      <p>You requested to reset your password.</p>
      <p>Click the button below to create a new password:</p>
      <a href="${resetUrl}" style="background: #4F46E5; color: white; padding: 12px 24px; text-decoration: none; border-radius: 6px;">
        Reset Password
      </a>
      <p>This link expires in 1 hour.</p>
      <p>If you didn't request this, please ignore this email.</p>
    `,
  },

  emailNotification: {
    subject: 'New notification from Soralia Village',
    getHtml: (title: string, message: string) => `
      <h1>${title}</h1>
      <p>${message}</p>
      <p>View on portal: https://soralia.co.za/notifications</p>
    `,
  },
};

export type TemplateKey = keyof typeof templates;
