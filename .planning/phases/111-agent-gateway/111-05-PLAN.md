<objective>
Wire maintenance request routing to the property occupancy model and the delegation system.

Three deliverables:

1. **Routing engine** — maintenance requests automatically route to HOA or landlord based on
   the property's active Household.occupancyType. RENTAL → landlord. OWNER_OCCUPIED/VACANT → HOA.

2. **ResidentDelegation API** — owners can grant renters the right to initiate specific platform
   actions (starting with maintenance:create). Lightweight — no token issuance, just a
   permission record checked at the maintenance create endpoint.

3. **Landlord workflow** — when a renter-raised request routes to a landlord, the landlord
   receives a notification, can acknowledge, and can delegate the work to a registered
   ServiceProvider with a scoped maintenance:coordinate grant.

HOA-routed requests are completely unchanged. This plan adds new paths alongside existing ones,
no existing maintenance functionality is modified destructively.
</objective>

```typescript
import { db } from '@api/server';
import type { MaintenanceRoutingContext } from './types';

/**
 * Resolve where a maintenance request should route, based on the
 * property's active Household occupancy type.
 *
 * Rules:
 *   RENTAL + known owner → LANDLORD (owner receives notification, handles request)
 *   OWNER_OCCUPIED        → HOA (owner is also resident; HOA manages)
 *   VACANT                → HOA (no tenant; HOA manages)
 *   No household found    → HOA (safe default)
 *   RENTAL + no owner     → HOA (cannot route to unknown landlord)
 */
export async function resolveRoutingType(
  propertyId: string,
  tenantId: string
): Promise<MaintenanceRoutingContext> {
  const property = await db.property.findFirst({
    where: { id: propertyId, tenantId },
    select: {
      ownerId: true,
      households: {
        where: { status: 'ACTIVE', deletedAt: null },
        orderBy: { createdAt: 'desc' },
        take: 1,
        select: { occupancyType: true },
      },
    },
  });

  if (!property) {
    return { routingType: 'HOA', landlordId: null, reason: 'property not found' };
  }

  const activeHousehold = property.households[0];

  if (!activeHousehold) {
    return { routingType: 'HOA', landlordId: null, reason: 'no active household' };
  }

  if (activeHousehold.occupancyType === 'RENTAL' && property.ownerId) {
    return {
      routingType: 'LANDLORD',
      landlordId: property.ownerId,
      reason: 'rental property with known owner',
    };
  }

  return {
    routingType: 'HOA',
    landlordId: null,
    reason: `occupancy type: ${activeHousehold.occupancyType}`,
  };
}
```

```typescript
import { NextRequest } from 'next/server';
import { z } from 'zod';
import { withTenant, getSessionAndRole, apiSuccess, apiError, db } from '@api/server';

export const maxDuration = 8;

/**
 * Scopes that an owner is permitted to delegate to a renter/occupant.
 * Subset of AGENT_SCOPES — renters cannot receive financials, documents:upload,
 * tenancy management, or listing scopes.
 */
const RENTER_DELEGATABLE_SCOPES = [
  'maintenance:create',
  'maintenance:read',
  'inspection:schedule',
  'inspection:view',
  'communication:notify_occupant',
] as const;

type RenterScope = (typeof RENTER_DELEGATABLE_SCOPES)[number];

const grantSchema = z.object({
  profileId: z.string().min(1),
  scopes: z.array(z.string().min(1)).min(1),
  expiresAt: z.string().datetime().optional(),
});

const revokeSchema = z.object({
  delegationId: z.string().min(1),
});

// POST — owner grants renter initiation rights
export async function POST(request: NextRequest, { params }: { params: { id: string } }) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const propertyId = params.id;

    // Verify caller owns the property
    const property = await db.property.findFirst({
      where: { id: propertyId, tenantId, ownerId: session.user.id },
    });
    if (!property) {
      return apiError('FORBIDDEN', 403, 'Only the property owner can grant resident delegations');
    }

    const body = await request.json();
    const parsed = grantSchema.safeParse(body);
    if (!parsed.success) {
      return apiError('VALIDATION_ERROR', 400, parsed.error.issues);
    }

    const { profileId, scopes, expiresAt } = parsed.data;

    // Validate scopes against renter-allowable subset
    const invalidScopes = scopes.filter(s => !RENTER_DELEGATABLE_SCOPES.includes(s as RenterScope));
    if (invalidScopes.length > 0) {
      return apiError(
        'VALIDATION_ERROR',
        400,
        `These scopes cannot be delegated to a renter: ${invalidScopes.join(', ')}. ` +
          `Allowed: ${RENTER_DELEGATABLE_SCOPES.join(', ')}`
      );
    }

    // Verify profile exists and is linked to this property's active household
    const profile = await db.profile.findFirst({
      where: {
        id: profileId,
        tenantId,
        household: { propertyId, status: 'ACTIVE' },
      },
    });
    if (!profile) {
      return apiError(
        'NOT_FOUND',
        404,
        'Profile not found or not an active occupant of this property'
      );
    }

    // Revoke any existing active delegations for this profile+property
    await db.residentDelegation.updateMany({
      where: { propertyId, profileId, revokedAt: null },
      data: { revokedAt: new Date() },
    });

    const delegation = await db.residentDelegation.create({
      data: {
        tenantId,
        propertyId,
        ownerId: session.user.id,
        profileId,
        scopes,
        expiresAt: expiresAt ? new Date(expiresAt) : null,
      },
    });

    return apiSuccess(
      {
        id: delegation.id,
        profileId,
        scopes,
        grantedAt: delegation.grantedAt.toISOString(),
        expiresAt: delegation.expiresAt?.toISOString() ?? null,
      },
      201
    );
  });
}

// GET — list active resident delegations for a property
export async function GET(request: NextRequest, { params }: { params: { id: string } }) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const propertyId = params.id;

    const property = await db.property.findFirst({
      where: { id: propertyId, tenantId },
    });
    if (!property) return apiError('NOT_FOUND', 404, 'Property not found');

    const isOwner = property.ownerId === session.user.id;
    const isAdmin = ['ADMIN', 'BOARD'].includes(session.user.role ?? '');
    if (!isOwner && !isAdmin) {
      return apiError('FORBIDDEN', 403, 'Access denied');
    }

    const delegations = await db.residentDelegation.findMany({
      where: { propertyId, tenantId, revokedAt: null },
      include: {
        profile: { select: { id: true, displayName: true, profileAddress: true } },
      },
      orderBy: { grantedAt: 'desc' },
    });

    return apiSuccess(
      delegations.map(d => ({
        id: d.id,
        profileId: d.profileId,
        profileName: d.profile.displayName,
        profileAddress: d.profile.profileAddress,
        scopes: d.scopes,
        grantedAt: d.grantedAt.toISOString(),
        expiresAt: d.expiresAt?.toISOString() ?? null,
      }))
    );
  });
}

// DELETE — owner revokes a resident delegation
export async function DELETE(request: NextRequest, { params }: { params: { id: string } }) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const body = await request.json();
    const parsed = revokeSchema.safeParse(body);
    if (!parsed.success) return apiError('VALIDATION_ERROR', 400, parsed.error.issues);

    const delegation = await db.residentDelegation.findFirst({
      where: {
        id: parsed.data.delegationId,
        propertyId: params.id,
        tenantId,
        revokedAt: null,
      },
      include: { property: { select: { ownerId: true } } },
    });

    if (!delegation) return apiError('NOT_FOUND', 404, 'Delegation not found');

    const isOwner = delegation.property.ownerId === session.user.id;
    const isAdmin = ['ADMIN', 'BOARD'].includes(session.user.role ?? '');
    if (!isOwner && !isAdmin) {
      return apiError('FORBIDDEN', 403, 'Only the property owner can revoke this delegation');
    }

    await db.residentDelegation.update({
      where: { id: parsed.data.delegationId },
      data: { revokedAt: new Date() },
    });

    return apiSuccess({ id: parsed.data.delegationId, revoked: true });
  });
}
```

```typescript
const [routingHint, setRoutingHint] = useState<'HOA' | 'LANDLORD' | null>(null);

// When propertyId changes, fetch the routing hint
useEffect(() => {

return { ..., routingHint };
```

Add a lightweight endpoint `src/app/api/maintenance/routing-hint/route.ts`:

```typescript
import { NextRequest } from 'next/server';
import { withTenant, getSessionAndRole, apiSuccess, apiError } from '@api/server';
import { resolveRoutingType } from '@entities/maintenance/server';

export async function GET(request: NextRequest) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const propertyId = new URL(request.url).searchParams.get('propertyId');
    if (!propertyId) return apiError('VALIDATION_ERROR', 400, 'propertyId required');

    const { routingType } = await resolveRoutingType(propertyId, tenantId);
    return apiSuccess({ routingType });
  });
}
```
