```typescript
export type DelegationActionType =
  | 'created'
  | 'accepted'
  | 'rejected'
  | 'revoked'
  | 'expired'
  | 'scope_modified'
  | 'token_issued';

/**
 * Log a delegation action to the DelegationAction audit trail.
 * Fire-and-forget unless caller awaits.
 */
export async function logDelegationAction(params: {
  tenantId: string;
  delegationId: string;
  action: DelegationActionType;
  actorId: string;
  metadata?: Record<string, unknown>;
}): Promise<void> {
  await db.delegationAction.create({
    data: {
      tenantId: params.tenantId,
      delegationId: params.delegationId,
      action: params.action,
      actorId: params.actorId,
      metadata: params.metadata ?? undefined,
    },
  });
}
```

```typescript
import { NextRequest } from 'next/server';
import { z } from 'zod';
import { withTenant, getSessionAndRole, apiSuccess, apiError, db } from '@api/server';
import { logDelegationAction } from '@api/shared/delegations';

export const maxDuration = 8;

const delegateSchema = z.object({
  providerId: z.string().min(1),
  scopes: z
    .array(
      z.enum([
        'VIEW_LISTING',
        'MANAGE_OCCUPANCY',
        'VIEW_FINANCIALS',
        'CONTACT_OCCUPANTS',
        'MARKET_PROPERTY',
      ])
    )
    .min(1)
    .max(5),
  expiresAt: z.string().datetime().optional(),
  contractTerms: z.string().max(2000).optional(),
});

export async function POST(request: NextRequest, { params }: { params: { id: string } }) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const propertyId = params.id;

    // Verify property exists and caller owns it
    const property = await db.property.findUnique({
      where: { id: propertyId, tenantId },
    });

    if (!property) {
      return apiError('NOT_FOUND', 404, 'Property not found');
    }

    // Owner check: the property must belong to the calling user
    // Property ownerId maps to the user who registered the property
    if (property.ownerId !== session.user.id) {
      return apiError('FORBIDDEN', 403, 'Only the property owner can delegate access');
    }

    const body = await request.json();
    const parsed = delegateSchema.safeParse(body);
    if (!parsed.success) {
      return apiError('VALIDATION_ERROR', 400, parsed.error.issues);
    }

    const { providerId, scopes, expiresAt, contractTerms } = parsed.data;

    // Verify provider exists
    const provider = await db.user.findUnique({
      where: { id: providerId },
    });

    if (!provider) {
      return apiError('NOT_FOUND', 404, 'Provider not found');
    }

    // Prevent self-delegation
    if (providerId === session.user.id) {
      return apiError('CONFLICT', 409, 'Cannot delegate to yourself');
    }

    // Check for existing active/pending delegation for same property+agent
    const existing = await db.agentAccess.findFirst({
      where: {
        tenantId,
        propertyId,
        agentId: providerId,
        status: { in: ['PENDING', 'ACTIVE'] },
      },
    });

    if (existing) {
      return apiError(
        'CONFLICT',
        409,
        'An active or pending delegation already exists for this property and provider'
      );
    }

    // Default expiry: 90 days from now if not specified
    const expiry = expiresAt
      ? new Date(expiresAt)
      : new Date(Date.now() + 90 * 24 * 60 * 60 * 1000);

    // Create the delegation (AgentAccess record)
    const delegation = await db.agentAccess.create({
      data: {
        tenantId,
        agentId: providerId,
        propertyId,
        grantedById: session.user.id,
        permissions: scopes,
        status: 'PENDING',
        startedAt: new Date(),
        expiresAt: expiry,
        contractTerms: contractTerms ?? null,
      },
    });

    // Audit: delegation created
    await logDelegationAction({
      tenantId,
      delegationId: delegation.id,
      action: 'created',
      actorId: session.user.id,
      metadata: {
        scopes,
        expiresAt: expiry.toISOString(),
      },
    });

    return apiSuccess(
      {
        id: delegation.id,
        propertyId,
        providerId,
        permissions: scopes,
        status: 'PENDING',
        expiresAt: expiry.toISOString(),
        createdAt: delegation.createdAt.toISOString(),
      },
      201
    );
  });
}
```

```typescript
import { NextRequest } from 'next/server';
import { withTenant, getSessionAndRole, apiSuccess, apiError, db } from '@api/server';
import { signAgentToken, hashToken } from '@api/lib/agent-token';
import { logDelegationAction } from '@api/shared/delegations';
import type { AgentScope } from '@entities/agent';
import { AgentPermission } from '@prisma/client';

export const maxDuration = 8;

export async function POST(request: NextRequest, { params }: { params: { id: string } }) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const delegationId = params.id;

    const delegation = await db.agentAccess.findUnique({
      where: { id: delegationId },
    });

    if (!delegation || delegation.tenantId !== tenantId) {
      return apiError('NOT_FOUND', 404, 'Delegation not found');
    }

    // Only the target provider can accept
    if (delegation.agentId !== session.user.id) {
      return apiError('FORBIDDEN', 403, 'Only the delegated provider can accept this delegation');
    }

    // Must be PENDING
    if (delegation.status !== 'PENDING') {
      return apiError(
        'CONFLICT',
        409,
        `Cannot accept a ${delegation.status.toLowerCase()} delegation`
      );
    }

    // Check expiry
    if (new Date(delegation.expiresAt) < new Date()) {
      // Auto-reject expired
      await db.agentAccess.update({
        where: { id: delegationId },
        data: { status: 'EXPIRED' },
      });
      await logDelegationAction({
        tenantId,
        delegationId,
        action: 'expired',
        actorId: 'system',
        metadata: { autoExpiredAt: new Date().toISOString() },
      });
      return apiError('GONE', 410, 'This delegation has expired');
    }

    // D-18: Provider verification gate
    const agentProfile = await db.agentProfile.findUnique({
      where: { agentId: session.user.id },
    });

    if (!agentProfile || !agentProfile.isVerified) {
      return apiError('FORBIDDEN', 403, 'Only verified providers can accept delegations');
    }

    // Build effective scope from delegation permissions
    const scope: AgentScope = {
      spaces: ['services'], // Provider dashboard space
      pages: delegation.permissions.map(p => p.toLowerCase()),
      apis: buildApiScopes(delegation.permissions),
      dataDomains: delegation.permissions.map(p => p.toLowerCase()),
      actions: ['read', ...getWriteActions(delegation.permissions)],
      maxDuration: delegation.expiresAt
        ? Math.ceil((new Date(delegation.expiresAt).getTime() - Date.now()) / 1000)
        : 86400 * 90,
    };

    // Update delegation status
    await db.agentAccess.update({
      where: { id: delegationId },
      data: {
        status: 'ACTIVE',
        acceptedAt: new Date(),
      },
    });

    // Issue an AgentToken linked to this delegation
    const tokenId = `dt_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
    const rawToken = await signAgentToken({
      agentId: delegation.agentId,
      tokenId,
      tenantId,
      callerType: 'delegated',
      scope,
      delegationId,
      expiresInSeconds: scope.maxDuration,
    });

    const tokenHash = hashToken(rawToken);

    await db.agentToken.create({
      data: {
        id: tokenId,
        tenantId,
        agentId: delegation.agentId,
        issuedById: session.user.id,
        accessId: delegationId,
        name: `Delegation token for property ${delegation.propertyId}`,
        tokenHash,
        scope: scope as unknown as Record<string, unknown>,
        expiresAt: new Date(Date.now() + scope.maxDuration * 1000),
      },
    });

    // Audit
    await logDelegationAction({
      tenantId,
      delegationId,
      action: 'accepted',
      actorId: session.user.id,
      metadata: { tokenId },
    });

    await logDelegationAction({
      tenantId,
      delegationId,
      action: 'token_issued',
      actorId: session.user.id,
      metadata: { tokenId },
    });

    return apiSuccess({
      id: delegationId,
      status: 'ACTIVE',
      token: rawToken,
      tokenId,
      expiresAt: new Date(Date.now() + scope.maxDuration * 1000).toISOString(),
    });
  });
}

function buildApiScopes(permissions: AgentPermission[]): string[] {
  const map: Record<string, string[]> = {
    VIEW_LISTING: ['properties:read'],
    MANAGE_OCCUPANCY: ['properties:write', 'maintenance:write'],
    VIEW_FINANCIALS: ['finances:read'],
    CONTACT_OCCUPANTS: ['messages:write'],
    MARKET_PROPERTY: ['marketplace:write'],
  };
  const apis = new Set<string>();
  for (const p of permissions) {
    (map[p] ?? []).forEach(a => apis.add(a));
  }
  return [...apis];
}

function getWriteActions(permissions: AgentPermission[]): string[] {
  const writeScopes = ['MANAGE_OCCUPANCY', 'CONTACT_OCCUPANTS', 'MARKET_PROPERTY'];
  const hasWrite = permissions.some(p => writeScopes.includes(p));
  return hasWrite ? ['write'] : [];
}
```

```typescript
import { NextRequest } from 'next/server';
import { withTenant, getSessionAndRole, apiSuccess, apiError, db } from '@api/server';
import { logDelegationAction } from '@api/shared/delegations';

export const maxDuration = 5;

export async function PATCH(request: NextRequest, { params }: { params: { id: string } }) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const delegation = await db.agentAccess.findUnique({
      where: { id: params.id },
    });

    if (!delegation || delegation.tenantId !== tenantId) {
      return apiError('NOT_FOUND', 404, 'Delegation not found');
    }

    // Owner (grantedById) OR admin can revoke
    const isOwner = delegation.grantedById === session.user.id;
    const isAdmin = ['ADMIN', 'BOARD'].includes(session.user.role ?? '');

    if (!isOwner && !isAdmin) {
      return apiError(
        'FORBIDDEN',
        403,
        'Only the property owner or admin can revoke this delegation'
      );
    }

    if (delegation.status !== 'ACTIVE' && delegation.status !== 'PENDING') {
      return apiError(
        'CONFLICT',
        409,
        `Cannot revoke a ${delegation.status.toLowerCase()} delegation`
      );
    }

    // Revoke the delegation
    await db.agentAccess.update({
      where: { id: params.id },
      data: { status: 'REVOKED', revokedAt: new Date() },
    });

    // Cascade: revoke all linked agent tokens
    await db.agentToken.updateMany({
      where: { accessId: params.id, revokedAt: null },
      data: { revokedAt: new Date() },
    });

    await logDelegationAction({
      tenantId,
      delegationId: params.id,
      action: 'revoked',
      actorId: session.user.id,
      metadata: { revokedBy: isAdmin ? 'admin' : 'owner' },
    });

    return apiSuccess({ id: params.id, status: 'REVOKED' });
  });
}
```

```typescript
import { NextRequest } from 'next/server';
import { withTenant, getSessionAndRole, apiSuccess, apiError, db } from '@api/server';

export const maxDuration = 5;

export async function GET(request: NextRequest) {
  return withTenant(async tenantId => {
    const session = await getSessionAndRole(request);
    if (!session) return apiError('UNAUTHORIZED', 401, 'Authentication required');

    const url = new URL(request.url);
    const propertyId = url.searchParams.get('propertyId');
    const status = url.searchParams.get('status');
    const agentId = url.searchParams.get('agentId');

    const where: Record<string, unknown> = { tenantId };

    // Filter by property
    if (propertyId) where.propertyId = propertyId;

    // Filter by status
    if (status) {
      const validStatuses = ['PENDING', 'ACTIVE', 'REJECTED', 'REVOKED', 'EXPIRED'];
      if (!validStatuses.includes(status.toUpperCase())) {
        return apiError(
          'VALIDATION_ERROR',
          400,
          `Invalid status: ${status}. Must be one of: ${validStatuses.join(', ')}`
        );
      }
      where.status = status.toUpperCase();
    }

    // Filter by agent
    if (agentId) where.agentId = agentId;

    // Owners see delegations they granted; providers see delegations assigned to them; admins see all
    const isAdmin = ['ADMIN', 'BOARD'].includes(session.user.role ?? '');
    if (!isAdmin) {
      where.OR = [{ grantedById: session.user.id }, { agentId: session.user.id }];
    }

    const delegations = await db.agentAccess.findMany({
      where: where as Parameters<typeof db.agentAccess.findMany>[0],
      include: {
        property: {
          select: { id: true, street: true, unit: true },
        },
        user_agentAccess_agentIdTouser: {
          select: { id: true, name: true, email: true },
        },
        user_agentAccess_grantedByIdTouser: {
          select: { id: true, name: true, email: true },
        },
      },
      orderBy: { createdAt: 'desc' },
      take: 100,
    });

    const formatted = delegations.map(d => ({
      id: d.id,
      propertyId: d.propertyId,
      propertyAddress: d.property
        ? `${d.property.street}${d.property.unit ? ` ${d.property.unit}` : ''}`
        : null,
      agentId: d.agentId,
      agentName: d.user_agentAccess_agentIdTouser?.name ?? null,
      agentEmail: d.user_agentAccess_agentIdTouser?.email ?? null,
      grantedById: d.grantedById,
      grantedByName: d.user_agentAccess_grantedByIdTouser?.name ?? null,
      permissions: d.permissions,
      status: d.status,
      startedAt: d.startedAt.toISOString(),
      expiresAt: d.expiresAt.toISOString(),
      acceptedAt: d.acceptedAt?.toISOString() ?? null,
      rejectedAt: d.rejectedAt?.toISOString() ?? null,
      createdAt: d.createdAt.toISOString(),
    }));

    return apiSuccess(formatted);
  });
}
```

## Trust Boundaries

| Boundary                                   | Description                                                                                             |
| ------------------------------------------ | ------------------------------------------------------------------------------------------------------- |
| owner → /api/properties/[id]/delegate      | Only property owner can create delegations. Self-delegation blocked. Duplicate delegation guarded.      |
| provider → /api/delegations/[id]/accept    | Only the target provider can accept. Provider must be verified (D-18). Expired delegations auto-reject. |
| owner/admin → /api/delegations/[id]/revoke | Owner or admin can revoke. Cascades to linked agent tokens (revokedAt set).                             |
| GET /api/delegations                       | Owners see their granted delegations; providers see delegations assigned to them; admins see all.       |

## STRIDE Threat Register

| Threat ID | Category               | Component                | Disposition | Mitigation Plan                                                                                                          |
| --------- | ---------------------- | ------------------------ | ----------- | ------------------------------------------------------------------------------------------------------------------------ |
| T-111-07  | Spoofing               | Delegate endpoint        | mitigate    | Property ownership verified by property.ownerId === session.user.id. No role-based spoofing possible.                    |
| T-111-08  | Elevation of Privilege | Accept endpoint          | mitigate    | agentId must match session.user.id. Provider verification (isVerified) enforced at acceptance.                           |
| T-111-09  | Tampering              | Revoke cascade           | mitigate    | Revoke sets REVOKED on AgentAccess AND revokedAt on all linked AgentTokens. No partial-revoke race condition.            |
