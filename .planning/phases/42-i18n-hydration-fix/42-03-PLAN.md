```tsx
import { useTranslation } from 'react-i18next';
const { t, ready } = useTranslation(['common', 'services']);

useEffect(() => {
  setMounted(true);
}, []);

};

import { useSafeTranslation } from '@features/i18n/model/useTranslation';

```
