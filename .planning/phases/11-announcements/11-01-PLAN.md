<objective>
Add audience targeting to the Announcement schema and build the CRUD API with notification fanout.
Purpose: Without targeting, announcements broadcast identically to all users regardless of role or occupancy type. The Group model already uses ResidentFilter for this exact purpose — announcements must adopt the same pattern. And without notification fanout, residents only see announcements by navigating to a dedicated page — a discoverability failure.
Output: Schema migration (targetFilter, targetRoles, updatedAt), Drizzle regeneration, API routes with targeting logic, notification fanout on POST.
</objective>

   ```ts
   // Step 1: Fetch all active users for this tenant
   let targetUsers = await db
     .select({ id: users.id })
     .from(users)
     .where(and(eq(users.tenantId, tenantId), eq(users.isActive, true)));

   // Step 2: Apply targetFilter
   if (targetFilter === 'OWNERS_ONLY') {
     // Join through profile where residencyType IN ('OWNER_RESIDENT', 'FAMILY')
     // OR user has a standardSeat record for this tenant
     const ownerUserIds = await db
       .select({ id: users.id })
       .from(users)
       .innerJoin(profiles, eq(profiles.userId, users.id))
       .where(
         and(
           eq(users.tenantId, tenantId),
           eq(users.isActive, true),
           inArray(profiles.residencyType, ['OWNER_RESIDENT', 'FAMILY'])
         )
       );
     targetUsers = ownerUserIds;
   } else if (targetFilter === 'RENTERS_ONLY') {
     const renterUserIds = await db
       .select({ id: users.id })
       .from(users)
       .innerJoin(profiles, eq(profiles.userId, users.id))
       .where(
         and(
           eq(users.tenantId, tenantId),
           eq(users.isActive, true),
           eq(profiles.residencyType, 'RENTER')
         )
       );
     targetUsers = renterUserIds;
   }
   // ALL: no occupancy filter — keep all active users

   // Step 3: Apply targetRoles (if non-empty)
   if (targetRoles.length > 0) {
     const roleFilteredUsers = await db
       .select({ id: users.id })
       .from(users)
       .where(
         and(
           eq(users.tenantId, tenantId),
           eq(users.isActive, true),
           inArray(users.role, targetRoles)
         )
       );

     // Intersect: user must match BOTH filter AND role
     const roleIds = new Set(roleFilteredUsers.map(u => u.id));
     targetUsers = targetUsers.filter(u => roleIds.has(u.id));
   }

   // Step 4: Cap at 500 and bulk insert notifications
   const cappedUsers = targetUsers.slice(0, 500);
   // TODO: Beyond 500 users, bulk job processing will be needed

   if (cappedUsers.length > 0) {
     await db.insert(notifications).values(
       cappedUsers.map(user => ({
         id: crypto.randomUUID(),
         tenantId,
         userId: user.id,
         title: announcement.title,
         message: announcement.content.slice(0, 200),
         type: 'announcement',
         link: `/news#announcement-${announcement.id}`,
         read: false,
       }))
     );
   }
   ```
