      }

        if (flagKey !== null) {
          expect(validFlagKeys.has(flagKey as string)).toBe(true);
        }
      }
    });

    it('should have exactly 14 entries (one per FeatureKey)', () => {
      expect(Object.keys(FEATURE_TO_FLAG)).toHaveLength(14);
    });
  });

  describe('FEATURE_TO_REGISTRY', () => {
    it('should have an entry for every FeatureKey', () => {
      for (const key of ALL_FEATURE_KEYS) {
        expect(FEATURE_TO_REGISTRY).toHaveProperty(key);
      }
    });

    it('registry keys should match page.<feature> format or be null', () => {

    });

      const registryKeys = Object.keys(FEATURE_TO_REGISTRY).sort();
      expect(flagKeys).toEqual(moduleKeys);
      expect(registryKeys).toEqual(moduleKeys);
    });

The file already has 7 helpers (`revalidateDashboard`, `revalidateDirectory`, `revalidateContent`, `revalidateConversations`, `revalidateAdminChanges`, `revalidateUserData`) — all using `revalidatePath()`. **Match this pattern. Do NOT introduce `revalidateTag()`.**

### Add the new helper at the end of the file:

```typescript
import { revalidatePath } from 'next/cache';

/**
 * Revalidate gate-related caches when tier/module/flag changes.
 * Called after:
 * - Tenant tier change (upgrade/downgrade)
 * - Module install/uninstall (onboarding or settings)
 * - Page flag toggle (admin settings UI)
 *
 * Uses revalidatePath() for consistency with the rest of this file.
 * Phase 1 is additive — this is the integration point for future
 * tier/module/flag mutation routes.
 *
 * @param tenantId - The tenant whose gate state changed (currently unused;
 *                   included for forward-compat with per-tenant caching)
 */
export function revalidateGate(tenantId: string): void {
  // Invalidate flag endpoint (carries platform flags used by Layer 3)
  revalidatePath('/api/flags');

  // Invalidate all gated page paths
  revalidatePath('/dashboard');
  revalidatePath('/admin');
  revalidatePath('/maintenance');
  revalidatePath('/bookings');
  revalidatePath('/events');
  revalidatePath('/surveys');
  revalidatePath('/competitions');
  revalidatePath('/directory');
  revalidatePath('/resources');
  revalidatePath('/chat');
  revalidatePath('/news');
  revalidatePath('/groups');
  revalidatePath('/services');
  revalidatePath('/messages');

  // Suppress unused parameter warning — used in Phase 2 for per-tenant targeting
  void tenantId;
}
```

**Design note:** This is conservative — invalidating more paths is safe; missing an invalidation would cause stale gate state. The function accepts `tenantId` for forward-compat (Phase 2 may add per-tenant invalidation via tagged caching).

### Integration points (not in Phase 1, but documented)

Where `revalidateGate()` should be called:

1. **Tier change** (`/api/admin/tenants/[id]/tier` route) — after updating `tenants.tier`
2. **Module install** (onboarding flow) — after enabling a module in `tenant_modules`
3. **Module uninstall** (settings page) — after disabling a module
4. **Page flag toggle** (admin settings UI) — after toggling a flag in `settings`

These integrations are deferred to Phase 2 (when the callsites are migrated to use `canAccess()`).

**Important notes:**

- Use the existing `revalidatePath()` pattern. Do NOT introduce `revalidateTag()`.
- The function takes `tenantId` even if the current implementation doesn't use it (Phase 2 may add per-tenant invalidation).
- Add the function at the end of the file (after `revalidateUserData`).
- Do NOT modify the existing 7 functions.
- Use `void tenantId;` to suppress the unused-parameter warning without changing the signature.
