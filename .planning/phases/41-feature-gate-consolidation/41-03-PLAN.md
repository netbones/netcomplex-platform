### 3. Section 1: Mapping Completeness (static)

The most important tests. Catch drift when a developer adds a `FeatureKey` but forgets to update a mapping.

```typescript
describe('Mapping completeness', () => {
  // List of all FeatureKey values (mirror the type, or derive from a shared constant)
  const ALL_FEATURE_KEYS: FeatureKey[] = [
    'maintenance',
    'bookings',
    'events',
    'surveys',
    'competitions',
    'groups',
    'chat',
    'news',
    'directory',
    'resources',
    'conservation',
    'services',
    'dashboard',
    'messages',
  ];

  describe('FEATURE_TO_MODULE', () => {
    it('should have an entry for every FeatureKey', () => {
      for (const key of ALL_FEATURE_KEYS) {
        expect(FEATURE_TO_MODULE).toHaveProperty(key);
        // Value can be a ModuleKey string or null
        expect([null, expect.any(String)]).toContainEqual(expect.objectContaining([key]));
      }
    });

    it('should only reference valid ModuleKey values or null', () => {
      const validModuleKeys = new Set(Object.keys(MODULES));
      for (const [feature, moduleKey] of Object.entries(FEATURE_TO_MODULE)) {
        if (moduleKey !== null) {
          expect(validModuleKeys.has(moduleKey as string)).toBe(true);
        }
      }
    });

    it('should have exactly 14 entries (one per FeatureKey)', () => {
      expect(Object.keys(FEATURE_TO_MODULE)).toHaveLength(14);
    });
  });

  describe('FEATURE_TO_FLAG', () => {
    it('should have an entry for every FeatureKey', () => {
      for (const key of ALL_FEATURE_KEYS) {
        expect(FEATURE_TO_FLAG).toHaveProperty(key);
      }
    });

    it('should only reference valid PlatformPageFlags keys or null', () => {
      // Use a sample PlatformPageFlags object to get keys
      const validFlagKeys = new Set([
        'campaign',
        'conservation',
        'conservationExternalUrl',
        'chat',
        'news',
        'events',
        'directory',
        'groups',
        'services',
        'resources',
        'maintenance',
        'surveys',
        'competitions',
        'dashboard',
        'bookings',
        'messages',
        'headerEngagementFocus',
      ]);
      for (const [feature, flagKey] of Object.entries(FEATURE_TO_FLAG)) {
        if (flagKey !== null) {
          expect(validFlagKeys.has(flagKey as string)).toBe(true);
        }
      }
    });

    it('should have exactly 14 entries (one per FeatureKey)', () => {
      expect(Object.keys(FEATURE_TO_FLAG)).toHaveLength(14);
    });
  });

  describe('FEATURE_TO_REGISTRY', () => {
    it('should have an entry for every FeatureKey', () => {
      for (const key of ALL_FEATURE_KEYS) {
        expect(FEATURE_TO_REGISTRY).toHaveProperty(key);
      }
    });

    it('should only reference valid FEATURE_REGISTRY keys or null', () => {
      // FEATURE_REGISTRY keys look like 'page.surveys', 'feature.x', 'widget.y'
      // For Phase 1, we accept any string that starts with 'page.', 'feature.', or 'widget.'
      for (const [feature, registryKey] of Object.entries(FEATURE_TO_REGISTRY)) {
        if (registryKey !== null) {
          expect(typeof registryKey).toBe('string');
          expect(registryKey as string).toMatch(/^(page|feature|widget)\./);
        }
      }
    });

    it('should have exactly 14 entries (one per FeatureKey)', () => {
      expect(Object.keys(FEATURE_TO_REGISTRY)).toHaveLength(14);
    });
  });

  describe('Cross-table consistency', () => {
    it('all 3 mapping tables have the same set of FeatureKey keys', () => {
      const moduleKeys = Object.keys(FEATURE_TO_MODULE).sort();
      const flagKeys = Object.keys(FEATURE_TO_FLAG).sort();
      const registryKeys = Object.keys(FEATURE_TO_REGISTRY).sort();
      expect(flagKeys).toEqual(moduleKeys);
      expect(registryKeys).toEqual(moduleKeys);
    });
  });
});
```

### 4. Section 2: Gate Function Behaviour (dynamic)

Test that `canAccess()` evaluates 5 layers in order, short-circuits on first `false`, and returns correct `GateResult`.

```typescript
describe('canAccess()', () => {
  const baseCtx: GateContext = {
    tenantId: 'tenant-1',
    userId: 'user-1',
    role: 'RESIDENT',
    tier: 'PREMIUM', // Old tier system: STANDARD | PREMIUM | ENTERPRISE
  };

  beforeEach(() => {
    vi.clearAllMocks();
  });

  it('should allow when all 5 layers pass', async () => {
    vi.mocked(isModuleEnabled).mockResolvedValue(true);
    vi.mocked(getPlatformPageFlags).mockResolvedValue({
      // All flags true
      campaign: true,
      conservation: 'default',
      conservationExternalUrl: '',
      chat: true,
      news: true,
      events: true,
      directory: true,
      groups: true,
      services: true,
      resources: true,
      maintenance: true,
      surveys: true,
      competitions: true,
      dashboard: true,
      bookings: true,
      messages: true,
      headerEngagementFocus: 'conservation',
    });

    const result = await canAccess(baseCtx, 'maintenance');
    expect(result).toEqual({ allowed: true, reason: 'allowed' });
  });

  it('should deny with reason="tier" when tenant tier is too low', async () => {
    const lowTierCtx = { ...baseCtx, tier: 'STANDARD' as const };
    // MODULES.maintenance.tier is 'depth' (assuming mapping) → STANDARD (foundation) < depth

    const result = await canAccess(lowTierCtx, 'maintenance');
    expect(result.allowed).toBe(false);
    expect(result.reason).toBe('tier');
  });

  it('should deny with reason="module" when isModuleEnabled returns false', async () => {
    vi.mocked(isModuleEnabled).mockResolvedValue(false);

    const result = await canAccess(baseCtx, 'maintenance');
    expect(result.allowed).toBe(false);
    expect(result.reason).toBe('module');
  });

  it('should deny with reason="flag" when page flag is false', async () => {
    vi.mocked(isModuleEnabled).mockResolvedValue(true);
    vi.mocked(getPlatformPageFlags).mockResolvedValue({
      // ... all defaults except maintenance: false
      maintenance: false,
      // ... (use spread with defaults)
    } as PlatformPageFlags);

    const result = await canAccess(baseCtx, 'maintenance');
    expect(result.allowed).toBe(false);
    expect(result.reason).toBe('flag');
  });

  it('should skip flag check when skipFlag=true', async () => {
    vi.mocked(isModuleEnabled).mockResolvedValue(true);
    vi.mocked(getPlatformPageFlags).mockResolvedValue({
      // All flags false
      maintenance: false,
      // ... (rest false)
    } as PlatformPageFlags);

    const result = await canAccess(baseCtx, 'maintenance', { skipFlag: true });
    expect(result.allowed).toBe(true);
  });

  it('should short-circuit on first failing layer (role before tier)', async () => {
    // Hard to test role denial in Phase 1 (returns true for all)
    // This test documents the short-circuit behavior
    const lowTierCtx = { ...baseCtx, tier: 'STANDARD' as const };

    const result = await canAccess(lowTierCtx, 'maintenance');
    expect(result.reason).toBe('tier');

    // isModuleEnabled should NOT be called (tier check failed first)
    expect(isModuleEnabled).not.toHaveBeenCalled();
  });

  it('should handle features with null mappings (e.g., competitions, dashboard)', async () => {
    // competitions: module=null, flag='competitions', registry=null
    // With flag=true and no module/registry check, should allow
    vi.mocked(isModuleEnabled).mockResolvedValue(true); // shouldn't be called
    vi.mocked(getPlatformPageFlags).mockResolvedValue({
      // ... all flags true
    } as PlatformPageFlags);

    const result = await canAccess(baseCtx, 'competitions');
    expect(result.allowed).toBe(true);
  });
});

describe('GATE_REASON_TO_ERROR', () => {
```

### Add new helper function:

```typescript
/**
 * Revalidate gate-related caches when tier/module/flag changes.
 * Called after:
 * - Tenant tier change (upgrade/downgrade)
 * - Module install/uninstall
 * - Page flag toggle (admin settings UI)
 *
 * @param tenantId - The tenant whose gate state changed
 */
export function revalidateGate(tenantId: string): void {
  // Invalidate tier + module + flag caches for this tenant
  revalidatePath(`/api/flags?tenantId=${tenantId}`);
  revalidatePath(`/api/flags`);
  revalidatePath(`/api/modules/enabled?tenantId=${tenantId}`);

  // Invalidate any pages that consume the gate
  revalidatePath('/dashboard');
  revalidatePath('/admin');
  revalidatePath('/maintenance');
  revalidatePath('/bookings');
  revalidatePath('/events');
  revalidatePath('/surveys');
  revalidatePath('/competitions');
  revalidatePath('/directory');
  revalidatePath('/resources');
  revalidatePath('/chat');
  revalidatePath('/news');
  revalidatePath('/groups');
  revalidatePath('/services');
  revalidatePath('/messages');
}
```

**Design note:** The `revalidateGate()` helper invalidates the standard set of gated page paths. This is conservative — invalidating more paths is safe; missing an invalidation would cause stale gate state. In Phase 2, this can be made more targeted (only invalidate the specific feature's paths).

### Integration points (not in Phase 1, but documented)

Where `revalidateGate()` should be called:

1. **Tier change** (`/api/admin/tenants/[id]/tier` route) — after updating `tenants.tier`
2. **Module install** (onboarding flow) — after enabling a module in `tenant_modules`
3. **Module uninstall** (settings page) — after disabling a module
4. **Page flag toggle** (admin settings UI) — after toggling a flag in `settings`

These integrations are deferred to Phase 2 (when the callsites are migrated to use `canAccess()`).

**Important notes:**

- Use the existing `revalidatePath()` pattern. Do NOT introduce a new cache mechanism.
- The function takes `tenantId` as a parameter even if the current implementation doesn't use it (e.g., `revalidatePath` doesn't support per-tenant paths). Phase 2 may add per-tenant invalidation via tags.
- Add the function at the end of the file (after `revalidateUserData`).
- Do NOT modify the existing functions.
