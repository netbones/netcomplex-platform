### 4. `canAccessClient()` sync function

Mirrors server `canAccess()` but uses pre-fetched data. Synchronous. Same `GateResult` shape:

```typescript
export function canAccessClient(
  ctx: ClientGateContext,
  feature: FeatureKey,
  opts?: { skipFlag?: boolean }
): GateResult {
  // Layer 0: Role
  if (!checkRoleClient(ctx.role, feature)) {
    return { allowed: false, reason: 'role' };
  }

  // Layer 1: Tier
  const moduleKey = FEATURE_TO_MODULE[feature];
  if (moduleKey) {
    if (!checkTierClient(ctx.tier, moduleKey)) {
      return { allowed: false, reason: 'tier' };
    }
  }

  // Layer 2: Module (skip — can't check from client; always assume true)
  // Server is source of truth for module state
  // If module disabled, server returns 403 on API calls; client just renders optimistically

  // Layer 3: PageFlag
  if (!opts?.skipFlag) {
    const flagKey = FEATURE_TO_FLAG_CLIENT[feature]; // client uses different mapping
    if (flagKey) {
      const flagValue = ctx.flags[flagKey as keyof PlatformPageFlags];
      const flagOk = flagValue === true;
      if (!flagOk) {
        return { allowed: false, reason: 'flag' };
      }
    }
  }

  // Layer 4: FeatureToggle
  const registryKey = FEATURE_TO_REGISTRY[feature];
  if (registryKey) {
    const featureDef = FEATURE_REGISTRY[registryKey];
    if (featureDef) {
      if (ctx.tier === 'foundation' && featureDef.tier !== 'foundation') {
        return { allowed: false, reason: 'feature' };
      }
      if (ctx.tier === 'depth' && featureDef.tier === 'core') {
        return { allowed: false, reason: 'feature' };
      }
    }
  }

  return { allowed: true, reason: 'allowed' };
}
```

**Note on Module layer (Layer 2):** Client cannot reliably check `isModuleEnabled()` because that requires DB. The server is the source of truth — if a module is disabled, the server returns 403 on API calls. Client-side, we **skip the Module check** and rely on the Flag layer (which is the runtime toggle that should align with module state in most cases).

If a developer needs module-level client check, they can call a dedicated `/api/modules/enabled` endpoint (future work). For Phase 1, this is acceptable: tier check (Layer 1) catches tier-locked modules, and flag check (Layer 3) catches admin-toggled features.

**Alternative:** Add a `modules: Record<ModuleKey, boolean>` field to `ClientGateContext` populated from `/api/flags` or a new endpoint. For Phase 1, **document this gap in a code comment** and skip — Phase 2 can add the endpoint if needed.

### 3. `GateGuardProps` interface

```typescript
interface GateGuardProps {
  /** Feature key (canonical FeatureKey, not 'page.*' string) */
  feature: FeatureKey;
  /** Content to render when access is allowed */
  children: ReactNode;
  /** Fallback content when access is denied (default: null) */
  fallback?: ReactNode;
  /** Render prop: receives GateResult, returns custom UI */
  render?: (props: { result: GateResult }) => ReactNode;
  /** Skip PageFlag layer (for admin pages) */
  skipFlag?: boolean;
  /** Show loading skeleton while context is loading */
  loadingFallback?: ReactNode;
}
```

### 4. `GateGuard` component

```typescript
export function GateGuard({
  feature,
  children,
  fallback = null,
  render,
  skipFlag,
  loadingFallback = null,
}: GateGuardProps) {
  const ctx = useGateContext();

  // Context still loading
  if (!ctx) {
    return <>{loadingFallback}</>;
  }

  const result = canAccessClient(ctx, feature, { skipFlag });

  // Render prop (most specific)
  if (render) {
    return <>{render({ result })}</>;
  }

  // Boolean conditional
  if (result.allowed) {
    return <>{children}</>;
  }

  // Fallback (typically null to hide)
  return <>{fallback}</>;
}
```
