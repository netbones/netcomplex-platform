### 9. `resolveGateContext()` helper

```typescript
export async function resolveGateContext(
  tenantId: string,
  request?: Request
): Promise<GateContext> {
  // Fetch tenant tier (DB column type)
  const [tenant] = await db.select().from(tenants).where(eq(tenants.id, tenantId)).limit(1);

  if (!tenant) {
    throw new Error(`Tenant ${tenantId} not found`);
  }

  // Fetch session + role via the canonical helper (Q2=A — real session, not hardcoded)
  const authData = await getSessionAndRole(request);
  const role = (authData?.role as Role) ?? 'RESIDENT';

  return {
    tenantId,
    role,
    tier: tenant.tier, // Already TenantTier from DB (no normalization boundary in Phase 1)
  };
}
```

**Notes:**

- **Q2=A:** Role is read from the real session via the existing `getSessionAndRole()` helper in `@api/auth-utils`. The role is typed as `string` upstream and cast to the `Role` type. If the session is unauthenticated, the role defaults to `'RESIDENT'` (preserves the conservative default of the existing helper).
- **`request` is optional** — the existing helper works with or without a `Request` (uses `headers()` from `next/headers` for server components, or `request.headers` for API routes). This matches the pattern used by every authenticated API route in the codebase (see `src/app/api/surveys/[id]/route.ts` and 30+ other routes).
- **The `getSessionAndRole()` call adds one DB query** (for the user role — it already does this for suspension check). This is acceptable for the gate path; the existing 200+ callsites pay the same cost.
- **No normalization boundary** — `tenant.tier` is `TenantTier` (DB enum) and flows directly into `GateContext.tier`. The conversion to `TierLevel` happens only at the FeatureToggle layer (Layer 4) where `FEATURE_REGISTRY.hasFeature()` requires `TierLevel`. This matches the Phase 1 decision to skip the `normalizeTier()` bridge.

### 11. `canAccess()` server function

```typescript
export async function canAccess(
  ctx: GateContext,
  feature: FeatureKey,
  opts?: { skipFlag?: boolean }
): Promise<GateResult> {
  // Layer 0: Role (synchronous, in-memory)
  const rolePermissions = ROLE_PERMISSIONS[ctx.role];
  if (!rolePermissions) {
    logDenial(ctx, feature, 'role');
    return { allowed: false, reason: 'role' };
  }

  // Layer 1: Tier (synchronous, in-memory against MODULES.tier)
  const moduleKey = FEATURE_TO_MODULE[feature];
  if (moduleKey !== null) {
    // MODULES[moduleKey].tier is TierLevel; we compare ctx.tier (TenantTier) via the same map as isModuleEnabled
    // We need to convert: STANDARD → foundation, PREMIUM → depth, ENTERPRISE → core
    const tierLevel = tenantTierToTierLevel(ctx.tier);
    const requiredTier = MODULES_REQUIRED_TIER[moduleKey]; // TierLevel
    if (!tierAtLeast(tierLevel, requiredTier)) {
      logDenial(ctx, feature, 'tier');
      return { allowed: false, reason: 'tier' };
    }

    // Layer 2: Module (DB call, cached via existing isModuleEnabled)
    const moduleOk = await isModuleEnabled(ctx.tenantId, moduleKey);
    if (!moduleOk) {
      logDenial(ctx, feature, 'module');
      return { allowed: false, reason: 'module' };
    }
  }

  // Layer 3: PageFlag (DB call, via getPlatformPageFlags)
  if (!opts?.skipFlag) {
    const flagKey = FEATURE_TO_FLAG[feature];
    if (flagKey !== null) {
      const flags = await getPlatformPageFlags(ctx.tenantId);
      // Special case: 'conservation' is a tri-state ('default' | 'managed' | 'external'), treat as enabled when not false
      const flagValue = flags[flagKey];
      const flagOk = typeof flagValue === 'boolean' ? flagValue : flagValue !== false;
      if (!flagOk) {
        logDenial(ctx, feature, 'flag');
        return { allowed: false, reason: 'flag' };
      }
    }
  }

  // Layer 4: FeatureToggle (synchronous against FEATURE_REGISTRY)
  const registryKey = FEATURE_TO_REGISTRY[feature];
  if (registryKey !== null) {
    const tierLevel = tenantTierToTierLevel(ctx.tier);
    // FEATURE_REGISTRY uses hasFeature(tier, featureKey) where tier is TierLevel
    // and canAccessPage(role, pageKey, tier, featureFlags)
    const featureOk =
      hasFeature(registryKey, tierLevel) &&
      canAccessPage(registryKey.replace(/^page\./, ''), tierLevel);
    if (!featureOk) {
      logDenial(ctx, feature, 'feature');
      return { allowed: false, reason: 'feature' };
    }
  }

  return { allowed: true, reason: 'allowed' };
}
```

### 12. Helpers

```typescript
import { TIERS, MODULES } from '@shared/lib/constants/tiers';

const TENANT_TIER_TO_LEVEL: Record<TenantTier, TierLevel> = {
  STANDARD: 'foundation',
  PREMIUM: 'depth',
  ENTERPRISE: 'core',
};

const TIER_LEVEL_ORDER: Record<TierLevel, number> = {
  foundation: 1,
  depth: 2,
  core: 3,
};

function tenantTierToTierLevel(tier: TenantTier): TierLevel {
  return TENANT_TIER_TO_LEVEL[tier] ?? 'foundation';
}

function tierAtLeast(tenant: TierLevel, required: TierLevel): boolean {
  return TIER_LEVEL_ORDER[tenant] >= TIER_LEVEL_ORDER[required];
}

// MODULES[moduleKey].tier is TierLevel — extract just the tier
const MODULES_REQUIRED_TIER: Record<ModuleKey, TierLevel> = Object.fromEntries(
  Object.entries(MODULES).map(([k, v]) => [k, v.tier])
) as Record<ModuleKey, TierLevel>;
```
