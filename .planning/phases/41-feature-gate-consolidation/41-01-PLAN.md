### 10. `canAccess()` server function

The core function. Evaluates 5 layers in order, short-circuits on first `false`. Returns `GateResult`.

```typescript
export async function canAccess(
  ctx: GateContext,
  feature: FeatureKey,
  opts?: { skipFlag?: boolean }
): Promise<GateResult> {
  // Layer 0: Role check (synchronous, in-memory)
  const roleAllowed = checkRole(ctx.role, feature);
  if (!roleAllowed) {
    logDenial(ctx, feature, 'role');
    return { allowed: false, reason: 'role' };
  }

  // Layer 1: Tier check (synchronous against MODULES.tier)
  const moduleKey = FEATURE_TO_MODULE[feature];
  if (moduleKey) {
    const tierOk = checkTier(ctx.tier, moduleKey);
    if (!tierOk) {
      logDenial(ctx, feature, 'tier');
      return { allowed: false, reason: 'tier' };
    }
  }

  // Layer 2: Module check (DB call, but cached via existing isModuleEnabled)
  if (moduleKey) {
    const moduleOk = await isModuleEnabled(ctx.tenantId, moduleKey);
    if (!moduleOk) {
      logDenial(ctx, feature, 'module');
      return { allowed: false, reason: 'module' };
    }
  }

  // Layer 3: PageFlag check (DB call, cached via getPlatformPageFlags)
  if (!opts?.skipFlag) {
    const flagKey = FEATURE_TO_FLAG[feature];
    if (flagKey) {
      const flags = await getPlatformPageFlags(ctx.tenantId);
      const flagValue = flags[flagKey];
      // Special case: 'conservation' is a tri-state ('default' | 'managed' | 'external')
      // Treat 'default' as enabled, anything else as enabled too (different access pattern)
      const flagOk = flagValue === true || (flagKey === 'conservation' && flagValue !== false);
      if (!flagOk) {
        logDenial(ctx, feature, 'flag');
        return { allowed: false, reason: 'flag' };
      }
    }
  }

  // Layer 4: FeatureToggle check (synchronous against FEATURE_REGISTRY)
  const registryKey = FEATURE_TO_REGISTRY[feature];
  if (registryKey) {
    const featureDef = FEATURE_REGISTRY[registryKey];
    if (featureDef) {
      // Convert TenantTier to TierLevel for comparison
      const tierLevel = tenantTierToTierLevel(ctx.tier);
      if (tierLevel === 'foundation' && featureDef.tier !== 'foundation') {
        logDenial(ctx, feature, 'feature');
        return { allowed: false, reason: 'feature' };
      }
      if (tierLevel === 'depth' && featureDef.tier === 'core') {
        logDenial(ctx, feature, 'feature');
        return { allowed: false, reason: 'feature' };
      }
    }
  }

  return { allowed: true, reason: 'allowed' };
}
```

### 11. `checkRole()` private helper

Synchronous role check. Some features require specific roles (e.g., admin-only features). For Phase 1, return `true` for all (let the feature/module/flag layers decide). **TODO Phase 2: implement proper role-feature mapping.**

### 12. `checkTier()` private helper

Synchronous tier check. Compares `TenantTier` against `MODULES[moduleKey].tier` (`TierLevel`). Use existing `TIER_LEVELS` mapping from `assert-module-enabled.ts` (or re-derive from `getTierLevel` in `require-module.ts`).

### 13. `tenantTierToTierLevel()` converter

Maps `TenantTier` (STANDARD/PREMIUM/ENTERPRISE) → `TierLevel` (foundation/depth/core):

- STANDARD → foundation
- PREMIUM → depth
- ENTERPRISE → core

This is the **bridge between the two tier systems**. Document as temporary — full unification is a separate workstream (out of Phase 1 scope).

### 14. `logDenial()` private helper

One log per `false` result using `createComponentLogger('gate')`:

```typescript
function logDenial(ctx: GateContext, feature: FeatureKey, reason: GateReason): void {
  if (reason === 'allowed') return;
  gateLogger.info({
    event: 'gate.denied',
    tenantId: ctx.tenantId,
    userId: ctx.userId,
    feature,
    reason,
    role: ctx.role,
    tier: ctx.tier,
  });
}
```
