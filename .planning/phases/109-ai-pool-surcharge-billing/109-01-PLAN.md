    ```typescript
    import { NextResponse } from 'next/server';
    import { db, apiNotFound, apiInternalError, tenantAiUsages, apiUnauthorized } from '@api/server';
    ```

    ```typescript
    // Try platform admin first
    const platformGuard = await requirePlatformAdmin(request);
    let tenantId: string;
    if (platformGuard) {
      // Not platform admin — try tenant-scoped access

      if (!auth) return apiUnauthorized();
    ```

## Trust Boundaries

| Boundary          | Description                                                                                                                              |
| ----------------- | ---------------------------------------------------------------------------------------------------------------------------------------- |
| Vercel Cron → API | POST /api/cron/ai-pool-rollover uses CRON_SECRET Bearer token — compromise of the secret allows unauthorized billing event creation      |
| Admin → API       | GET /api/admin/platform/billing/invoices/[id]/pdf requires platform admin or tenant auth — unauthorized access could leak financial data |
| PDF generation    | buildOverageInvoicePdf() receives raw financial data — must not include API keys or secrets in PDF content                               |

| Threat ID | Category               | Component                    | Disposition | Mitigation Plan                                                                                                                           |
| --------- | ---------------------- | ---------------------------- | ----------- | ----------------------------------------------------------------------------------------------------------------------------------------- |

| T-109-02  | Information Disclosure | invoices/[id]/pdf route      | mitigate    | requirePlatformAdmin guard + tenant-scoped fallback via withTenant(). Tenants can only access their own invoices.                         |
| T-109-03  | Tampering              | buildOverageInvoicePdf()     | mitigate    | PDF generated server-side — client cannot tamper with content. All data sourced from DB records, not request body.                        |
| T-109-04  | Repudiation            | ensureOverageInvoiceRecord() | mitigate    | Immutable BillingEvent (AI_OVERAGE_CHARGED) recorded per tenant per billing month for audit trail.                                        |
