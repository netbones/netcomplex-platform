// One definition per setting. The key is the DB key string.
// From this we generate: SETTINGS_KEYS, SettingValueMap type,
// SETTINGS_VALUE_SCHEMAS, DEFAULT_PAGE_FLAGS, FLAG_DEFS.

interface SettingDef {
  dbKey: string;                    // e.g. 'page_campaign_enabled'
  flagKey?: keyof PlatformPageFlags; // e.g. 'campaign' (if this setting maps to a page flag)
  type: 'boolean' | 'enum' | 'json' | 'string';
  enumValues?: readonly string[];   // for enum type
  defaultValue: unknown;            // typed default
  description?: string;
}

// Master list — add new settings here
const SETTING_DEFS: SettingDef[] = [
  // Page flags (boolean)
  { dbKey: 'page_campaign_enabled', flagKey: 'campaign', type: 'boolean', defaultValue: true },
  { dbKey: 'page_chat_enabled', flagKey: 'chat', type: 'boolean', defaultValue: true },
  // ... all 22 page flag booleans ...

  // Conservation (enum)
  { dbKey: 'page_conservation_mode', flagKey: 'conservation', type: 'enum',
    enumValues: ['default','managed','external'], defaultValue: 'default' },
  { dbKey: 'page_conservation_external_url', flagKey: 'conservationExternalUrl',
    type: 'string', defaultValue: '' },

  // Header links (json array)
  { dbKey: 'header_links', flagKey: 'headerLinks', type: 'json', defaultValue: ['directory','groups','services','resources'] },

  // Non-flag settings
  { dbKey: 'services_config', type: 'json', defaultValue: null },
  { dbKey: 'provider_registration_mode', type: 'enum',
    enumValues: ['OPEN','INVITATION_ONLY'], defaultValue: 'INVITATION_ONLY' },
  { dbKey: 'interest_categories', type: 'json', defaultValue: [] },
  { dbKey: 'merit_tier_thresholds', type: 'json', defaultValue: {} },
  { dbKey: 'merit_expiry_days', type: 'string', defaultValue: '' },
  { dbKey: 'translation_provider', type: 'string', defaultValue: '' },
  { dbKey: 'translation_api_key', type: 'string', defaultValue: '' },
  { dbKey: 'custom_pages', type: 'json', defaultValue: null },
  { dbKey: 'custom_nav', type: 'json', defaultValue: null },
  { dbKey: 'stats_homes', type: 'string', defaultValue: '180' },
  { dbKey: 'stats_years', type: 'string', defaultValue: '30' },
  { dbKey: 'stats_bird_species', type: 'string', defaultValue: '0' },
  { dbKey: 'stats_native_plants', type: 'string', defaultValue: '0' },
];

// ── Generated from SETTING_DEFS ──

export const SETTINGS_KEYS = Object.freeze(
  Object.fromEntries(SETTING_DEFS.map(d => [d.dbKey.toUpperCase(), d.dbKey]))
) as Record<string, string>;

export type SettingKey = (typeof SETTINGS_KEYS)[keyof typeof SETTINGS_KEYS];

// SettingValueMap type
export type SettingValueMap = {
  [D in (typeof SETTING_DEFS)[number] as D['dbKey']]: D['defaultValue'];
};

// Zod validator map
export const SETTINGS_VALUE_SCHEMAS = Object.fromEntries(
  SETTING_DEFS.map(d => {
    if (d.type === 'boolean') return [d.dbKey, z.enum(['true','false'])];
    if (d.type === 'enum' && d.enumValues) return [d.dbKey, z.enum(d.enumValues)];
    if (d.type === 'json') return [d.dbKey, jsonStringSchema()];
    return [d.dbKey, z.string()];
  })
);

// FLAG_DEFS (page flag → db key mapping)
export const FLAG_DEFS = Object.fromEntries(
  SETTING_DEFS.filter(d => d.flagKey).map(d => [d.flagKey!, { settingKey: d.dbKey, type: d.type, enumValues: d.enumValues }])
);

// Default page flags
export const DEFAULT_PAGE_FLAGS: PlatformPageFlags = { ... };
// Generated from SETTING_DEFS entries that have flagKey

// Validation function
export function validateSettingValue(key: string, value: string): { valid: boolean; error?: string } { ... }

// Typed setting getter
export function getTypedSetting<K extends keyof SettingValueMap>(key: K, value: string): SettingValueMap[K] { ... }
