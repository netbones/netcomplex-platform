    ```typescript
    import { NextRequest } from 'next/server';
    import { auth, db, apiSuccess, apiError, apiUnauthorized, apiNotFound, apiInternalError } from '@api/server';
    import { withTenant } from '@entities/tenant/server';
    import { logError } from '@shared/lib';
    import { eq } from 'drizzle-orm';
    import { serviceBookings } from '@db/schema/service-bookings';
    import { checkoutRequestSchema } from '@entities/marketplace/schema';
    import { initializeCheckout } from '@entities/marketplace/api/checkout';

    export const maxDuration = 8;

    export async function POST(request: NextRequest) {
      try {
        // 1. Auth check
        const session = await auth.api.getSession({ headers: request.headers });
        if (!session?.user?.id) return apiUnauthorized();

        // 2. Parse and validate body
        const body = await request.json();
        const parsed = checkoutRequestSchema.safeParse(body);
        if (!parsed.success) return apiError('VALIDATION_ERROR', parsed.error.message, 400);

        // 3. Tenant isolation
        const { tenantId } = await withTenant();

        // 4. Fetch booking + verify ownership
        const [booking] = await db.select().from(serviceBookings).where(eq(serviceBookings.id, parsed.data.bookingId)).limit(1);
        if (!booking) return apiNotFound('Booking not found');
        if (booking.userId !== session.user.id) return apiError('FORBIDDEN', 403, 'Not your booking');
        if (booking.tenantId !== tenantId) return apiNotFound('Booking not found');

        // 5. Verify booking is in PENDING_CONFIRMATION status
        if (booking.status !== 'PENDING_CONFIRMATION') return apiError('VALIDATION_ERROR', 'Booking is not in a payable state', 400);

        // 6. Fetch listing (for price verification)
        // 7. Initialize checkout
        const result = await initializeCheckout({ booking, ... });

        // 8. Return payment URL
        if (result.status === 'configuration_required') return apiError('VALIDATION_ERROR', result.message, 400);
        return apiSuccess({ paymentUrl: result.paymentUrl, reference: result.reference });
      } catch (error) {
        logError({ component: 'checkout-api', operation: 'POST' }, 'Checkout error', error);
        return apiInternalError();
      }
    }
    ```

    Follow the auth pattern from inquiries/route.ts but use raw body parsing for signature verification:

    ```typescript
    import { NextRequest } from 'next/server';
    import { db, apiSuccess, apiError, apiInternalError } from '@api/server';
    import { PaystackService } from '@/server/payments/paystack';
    import { eq } from 'drizzle-orm';
    import { serviceBookings } from '@db/schema/service-bookings';
    import { paymentTransactions } from '@db/schema/payment-transactions';
    import { logError } from '@shared/lib';
    import { notifyPaymentReceived } from '@entities/marketplace/api/notification-triggers';

    const paystack = new PaystackService();

    export async function POST(request: NextRequest) {
      try {
        // 1. Read raw body for signature verification
        const rawBody = await request.text();
        const signature = request.headers.get('x-paystack-signature');

        if (!signature) return apiError('VALIDATION_ERROR', 400, 'Missing signature');

        // 2. Verify Paystack HMAC-SHA512 signature
        const isValid = paystack.verifyWebhookSignature(rawBody, signature);
        if (!isValid) return apiError('VALIDATION_ERROR', 400, 'Invalid signature');

        // 3. Parse event
        const event = JSON.parse(rawBody);
        // event.event === 'charge.success'
        // event.data.reference === 'svc-{bookingId}'
        // event.data.status === 'success'

        // 4. Extract bookingId from reference
        const reference = event.data.reference;
        if (!reference?.startsWith('svc-')) return apiError('VALIDATION_ERROR', 400, 'Invalid reference');

        const bookingId = reference.replace('svc-', '');

        // 5. Fetch booking
        const [booking] = await db.select().from(serviceBookings).where(eq(serviceBookings.id, bookingId)).limit(1);
        if (!booking) return apiError('NOT_FOUND', 404, 'Booking not found');

        // 6. Idempotency: if already CONFIRMED, return 200 without changes
        if (booking.status === 'CONFIRMED') return apiSuccess({ status: 'already_processed' });

        // 7. Update booking based on event
        if (event.event === 'charge.success') {
          await db.update(serviceBookings)
            .set({ status: 'CONFIRMED', paymentStatus: 'COMPLETED', updatedAt: new Date() })
            .where(eq(serviceBookings.id, bookingId));

          // 8. Update payment transaction
          await db.update(paymentTransactions)
            .set({ status: 'COMPLETED' })
            .where(eq(paymentTransactions.reference, reference));

          // 9. Fire notification (non-blocking)
          notifyPaymentReceived({ tenantId: booking.tenantId, providerId: booking.providerId, listingId: booking.listingId, listingTitle: 'from db', amount: Number(booking.price), transactionId: event.data.id }).catch(() => {});
        }

        return apiSuccess({ status: 'processed' });
      } catch (error) {
        logError({ component: 'marketplace-webhook', operation: 'POST' }, 'Webhook error', error);
        return apiInternalError();
      }
    }
    ```

    **Security hardening** (per RESEARCH.md threat patterns):
    - Signature verification is MANDATORY — never skip
    - Idempotency check prevents double-spend (check status before update)
    - Reference format validated (`svc-` prefix) before DB query
    - All errors return generic 400/404 — no internal details leaked
    - Notification fire-and-forget (`.catch(() => {})`) — never blocks webhook response
