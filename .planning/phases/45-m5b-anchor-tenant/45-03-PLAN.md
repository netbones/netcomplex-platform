**Step A — Create StandingBadge component (src/entities/directory/ui/StandingBadge.tsx):**
"use client" component.

```typescript
import { ShieldCheck, Shield, AlertTriangle } from 'lucide-react';
import { getStandingTier, type StandingTier } from '@shared/lib/merits-helpers';

interface StandingTierConfig {
  label: string;
  icon: typeof ShieldCheck;
  colorClasses: string;  // Tailwind classes for badge background + text
}

const STANDING_TIER_CONFIG: Record<StandingTier, StandingTierConfig> = {
  GOLD:   { label: 'Gold Standing', icon: ShieldCheck,     colorClasses: 'bg-amber-100 text-amber-800' },
  SILVER: { label: 'Silver Standing', icon: Shield,        colorClasses: 'bg-gray-200 text-gray-700' },
  BRONZE: { label: 'Bronze Standing', icon: Shield,        colorClasses: 'bg-orange-100 text-orange-800' },
  PROBATION: { label: 'Probation',   icon: AlertTriangle,  colorClasses: 'bg-red-100 text-red-800' },
};

export function getStandingTierConfig(points: number | null | undefined): StandingTierConfig | null {
  if (points == null) return null;
  const tier = getStandingTier(points);
  return STANDING_TIER_CONFIG[tier];
}

export function StandingBadge({ points, size = 'sm' }: { points: number | null | undefined; size?: 'sm' | 'md' }) {
  if (points == null) return null;
  const config = getStandingTierConfig(points);
  if (!config) return null;
  const Icon = config.icon;
  const sizeClasses = size === 'md' ? 'px-3 py-1 text-sm gap-2' : 'px-2 py-0.5 text-xs gap-1';
  const iconSize = size === 'md' ? 'w-4 h-4' : 'w-3 h-3';

  return (
    <span className={`inline-flex items-center rounded-full font-medium ${sizeClasses} ${config.colorClasses}`}>
      <Icon className={iconSize} aria-hidden="true" />
      {config.label}
    </span>
  );
}
```

Use exact color classes from UI-SPEC Standing Tier Badge Colors table — never deviate.
