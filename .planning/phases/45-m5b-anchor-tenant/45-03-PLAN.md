**Step A — Create StandingBadge (src/entities/directory/ui/StandingBadge.tsx):**

```typescript
import { ShieldCheck, Shield, Users, AlertTriangle } from 'lucide-react';
import { getStandingTier, type StandingTier } from '@shared/lib/merits-helpers';

interface StandingTierConfig {
  label: string;
  publicLabel: string | null;  // null = hidden from public view
  icon: typeof ShieldCheck;
  colorClasses: string;
}

const STANDING_TIER_CONFIG: Record<StandingTier, StandingTierConfig> = {
  GOLD:      { label: 'Gold Standing',     publicLabel: 'Gold Standing',     icon: ShieldCheck,    colorClasses: 'bg-amber-100 text-amber-800' },
  SILVER:    { label: 'Silver Standing',   publicLabel: 'Silver Standing',   icon: Shield,         colorClasses: 'bg-gray-200 text-gray-700' },
  BRONZE:    { label: 'Bronze Standing',   publicLabel: 'Community Member',  icon: Users,          colorClasses: 'bg-emerald-100 text-emerald-800' },
  WATCHLIST: { label: 'Watchlist',         publicLabel: null,                icon: AlertTriangle,  colorClasses: 'bg-yellow-100 text-yellow-800' },
  PROBATION: { label: 'Probation',         publicLabel: null,                icon: AlertTriangle,  colorClasses: 'bg-red-100 text-red-800' },
};

export function getStandingTierConfig(points: number | null | undefined, context: 'public' | 'admin' = 'public'): StandingTierConfig | null {
  if (points == null) return null;
  const tier = getStandingTier(points);
  const config = STANDING_TIER_CONFIG[tier];
  if (context === 'public' && config.publicLabel === null) return null;  // hide negative tiers from public
  return config;
}

export function StandingBadge({ points, context = 'public', size = 'sm' }: {
  points: number | null | undefined;
  context?: 'public' | 'admin';
  size?: 'sm' | 'md';
}) {
  const config = getStandingTierConfig(points, context);
  if (!config) return null;  // null for hidden tiers or null points
  const Icon = config.icon;
  const displayLabel = context === 'public' && config.publicLabel ? config.publicLabel : config.label;
  const sizeClasses = size === 'md' ? 'px-3 py-1 text-sm gap-2' : 'px-2 py-0.5 text-xs gap-1';
  const iconSize = size === 'md' ? 'w-4 h-4' : 'w-3 h-3';

  return (
    <span className={`inline-flex items-center rounded-full font-medium ${sizeClasses} ${config.colorClasses}`}>
      <Icon className={iconSize} aria-hidden="true" />
      {displayLabel}
    </span>
  );
}
```
