Create React hook for real-time message subscription:

```typescript
import { useEffect } from 'react';
import { createBrowserClient } from '@/lib/supabase';
import type { Message } from '@/lib/supabase';

export function useRealtimeMessages(
  conversationId: string,
  onNewMessage: (message: Message) => void
) {
  const supabase = createBrowserClient();

  useEffect(() => {
    // Subscribe to new messages in conversation
    const channel = supabase
      .channel(`messages:${conversationId}`)
      .on(
        'postgres_changes',
        {
          event: 'INSERT',
          schema: 'public',
          table: 'Message',
          filter: `conversationId=eq.${conversationId}`,
        },
        payload => {
          onNewMessage(payload.new as Message);
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [conversationId, onNewMessage]);
}
```

```typescript
import { useCallback, useRef } from 'react';
import { createBrowserClient } from '@/lib/supabase';

export function useTypingIndicator(conversationId: string, userId: string) {
  const supabase = createBrowserClient();
  const timeoutRef = useRef<NodeJS.Timeout>();

  const setTyping = useCallback(
    async (isTyping: boolean) => {
      // Broadcast to channel
      await supabase.channel(`typing:${conversationId}`).track({
        user_id: userId,
        is_typing: isTyping,
        updated_at: new Date().toISOString(),
      });
    },
    [conversationId, userId]
  );

  const startTyping = useCallback(() => {
    setTyping(true);
    // Clear existing timeout
    if (timeoutRef.current) clearTimeout(timeoutRef.current);
    // Set timeout to stop typing after 3 seconds
    timeoutRef.current = setTimeout(() => setTyping(false), 3000);
  }, [setTyping]);

  return { startTyping, setTyping };
}
```

```typescript
import { useEffect, useState } from 'react';
import { createBrowserClient } from '@/lib/supabase';

export function usePresence(conversationId: string, userId: string) {
  const supabase = createBrowserClient();
  const [onlineUsers, setOnlineUsers] = useState<string[]>([]);

  useEffect(() => {
    const channel = supabase.channel(`presence:${conversationId}`);

    channel
      .on('presence', { event: 'sync' }, () => {
        const state = channel.presenceState();
        const users = Object.keys(state).filter(u => u !== userId);
        setOnlineUsers(users);
      })
      .subscribe(async () => {
        await channel.track({ user_id: userId, online_at: new Date().toISOString() });
      });

    return () => {
      channel.unsubscribe();
    };
  }, [conversationId, userId]);

  return { onlineUsers };
}
```
