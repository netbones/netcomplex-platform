**Route 1: `snapshot/route.ts`** — GET
Returns the tenant's billing snapshot: current subscription, recent invoices, recent payments, available plans.

```typescript
import { NextRequest } from 'next/server';
import { withTenant } from '@entities/tenant/server';
import { getTenantBillingSnapshot } from '@shared/api/tenant-billing';
import { apiSuccess, apiInternalError, logError } from '@api/server';

export const maxDuration = 8;

export async function GET(request: NextRequest) {
  try {
    const { tenantId } = await withTenant(request);
    const snapshot = await getTenantBillingSnapshot(tenantId);
    return apiSuccess(snapshot);
  } catch (error) {
    logError(
      { component: 'tenant-billing-snapshot-api', operation: 'GET' },
      'Failed to get billing snapshot',
      error
    );
    return apiInternalError();
  }
}
```

**Route 2: `checkout/route.ts`** — POST
Initiates a subscription checkout. Accepts JSON body: `{ planId, paymentGateway, billingEmail, couponCode?, callbackUrl? }`.

```typescript
import { NextRequest } from 'next/server';
import { withTenant } from '@entities/tenant/server';
import { createTenantSubscriptionCheckout } from '@shared/api/tenant-billing';
import { apiSuccess, apiError, apiInternalError, logError } from '@api/server';
import { auth } from '@api/server';

export const maxDuration = 8;

export async function POST(request: NextRequest) {
  try {
    const { tenantId } = await withTenant(request);
    const body = await request.json();
    const session = await auth.api.getSession({ headers: request.headers });
    const billingEmail = session?.user?.email;

    if (!billingEmail) {
      return apiError('UNAUTHORIZED', 401, 'User email required for billing');
    }

    const result = await createTenantSubscriptionCheckout({
      tenantId,
      planId: body.planId,
      paymentGateway: body.paymentGateway || 'PAYSTACK',
      billingEmail,
      couponCode: body.couponCode,
      callbackUrl: body.callbackUrl || `${request.nextUrl.origin}/tenant/billing`,
    });

    if (!result.ok) {
      return apiError('VALIDATION_ERROR', result.status, result.message);
    }

    return apiSuccess(result.data);
  } catch (error) {
    logError(
      { component: 'tenant-billing-checkout-api', operation: 'POST' },
      'Checkout failed',
      error
    );
    return apiInternalError();
  }
}
```

**Route 3: `invoices/route.ts`** — GET
Returns the tenant's invoice list. Optional `?cursor=id&limit=20` for pagination.

```typescript
import { NextRequest } from 'next/server';
import { withTenant } from '@entities/tenant/server';
import { db } from '@api/server';
import { tenantInvoices } from '@schema/tenant-invoice';
import { eq, desc, lt } from 'drizzle-orm';
import { apiSuccess, apiInternalError, logError } from '@api/server';

export const maxDuration = 8;

export async function GET(request: NextRequest) {
  try {
    const { tenantId } = await withTenant(request);
    const { searchParams } = new URL(request.url);
    const cursor = searchParams.get('cursor');
    const limit = Math.min(parseInt(searchParams.get('limit') || '20'), 100);

    const invoices = await db
      .select()
      .from(tenantInvoices)
      .where(
        and(
          eq(tenantInvoices.tenantId, tenantId),
          cursor ? lt(tenantInvoices.id, cursor) : undefined
        )
      )
      .orderBy(desc(tenantInvoices.createdAt))
      .limit(limit);

    return apiSuccess(invoices);
  } catch (error) {
    logError(
      { component: 'tenant-billing-invoices-api', operation: 'GET' },
      'Failed to list invoices',
      error
    );
    return apiInternalError();
  }
}
```

import { useQuery } from '@tanstack/react-query';
import { BillingOverview, PlanSelector } from '@features/billing';

export default function BillingPage() {
